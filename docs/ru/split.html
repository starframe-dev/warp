split

split.go реализует основную структуру дерева макета для терминала Warp. Он определяет иерархию узлов для разделения панелей, поддерживает вертикальное/горизонтальное разделение, гибкие макеты и сворачиваемые панели.

CollapsedSize()

Возвращает свернутый размер для данного направления, или 0, если не свернут.

func (n *Node) CollapsedSize(d Direction) int
func (n *Node) CollapsedSize(d Direction) int {
    if n.Collapse == nil || !n.Collapse.Active {
        return 0
    }
    if d == Vertical {
        if n.Collapse.Width > 0 {
            return n.Collapse.Width
        }
        return 1
    }
    if n.Collapse.Height > 0 {
        return n.Collapse.Height
    }
    return 1
}

findNode()

Локализует узел, содержащий данную панель в дереве. Возвращает nil, если не найден.

func (n *Node) findNode(panel Panel) *Node
func (n *Node) findNode(panel Panel) *Node {
    if n == nil {
        return nil
    }
    if n.Panel == panel {
        return n
    }
    if n.Split != nil {
        if found := n.Split.First.findNode(panel); found != nil {
            return found
        }
        if found := n.Split.Second.findNode(panel); found != nil {
            return found
        }
    }
    if n.Flex != nil {
        for _, item := range n.Flex.Items {
            if found := item.Node.findNode(panel); found != nil {
                return found
            }
        }
    }
    return nil
}

replaceNode()

Заменяет oldNode на newNode в дереве. Возвращает true, если замена была успешна.

func (n *Node) replaceNode(old, new *Node) bool
func (n *Node) replaceNode(old, new *Node) bool {
    if n == nil {
        return false
    }
    if n.Split != nil {
        if n.Split.First == old {
            n.Split.First = new
            return true
        }
        if n.Split.Second == old {
            n.Split.Second = new
            return true
        }
        return n.Split.First.replaceNode(old, new) || n.Split.Second.replaceNode(old, new)
    }
    if n.Flex != nil {
        for _, item := range n.Flex.Items {
            if item.Node == old {
                item.Node = new
                return true
            }
            if item.Node.replaceNode(old, new) {
                return true
            }
        }
    }
    return false
}

collectLeafNodes()

Собирает все листовые (Panel) узлы по порядку.

func (n *Node) collectLeafNodes() []*Node
func (n *Node) collectLeafNodes() []*Node {
    if n == nil {
        return nil
    }
    if n.IsLeaf() {
        return []*Node{n}
    }
    var result []*Node
    if n.Split != nil {
        result = append(result, n.Split.First.collectLeafNodes()...)
        result = append(result, n.Split.Second.collectLeafNodes()...)
    }
    if n.Flex != nil {
        for _, item := range n.Flex.Items {
            result = append(result, item.Node.collectLeafNodes()...)
        }
    }
    return result
}

Иерархия макета

Дерево макета поддерживает три типа узлов:

  • Panel — Листовой узел терминала (фактическое содержимое окна)
  • Split — Делит пространство между двумя детьми (вертикально или горизонтально)
  • Flex — Распределяет детей с взвешенными размерами (строка или столбец)

Узлы могут быть свернуты, что фиксирует их размер в данном направлении.
