Обзор

Этот файл намеренно минимален и служит в качестве заглушки документации. Он явно документирует тот факт, что взаимодействия drag-and-drop для разделительных границ в этом файле НЕ реализуются.

Фактическая функциональность drag-and-drop для разделительных границ warp layout реализована в других файлах:

  • tab.go — Реализует обработку ввода и управление состоянием для перетаскивания границы между разделёнными панелями. Соответствующие функции — handleMouse, updateDrag и hitBorder.
  • render.go — Вычисляет координаты границ на экране посредством findBorders.

func (t *Tab) hitBorder(bh BorderHit, mx, my int) bool {
    switch bh.Direction {
    case Vertical:
        return mx == bh.X && my >= bh.Y && my < bh.Y+bh.Length
    case Horizontal:
        return my == bh.Y && mx >= bh.X && mx < bh.X+bh.Length
    }
    return false
}

func (t *Tab) updateDrag(mx, my, cw, ch int) {
    if t.dragging != nil {
        t.updateSplitDrag(mx, my, cw, ch)
    }
    if t.flexDragging != nil {
        t.updateFlexDrag(mx, my, cw, ch)
    }
    // Send live ResizeMsg so panels update while dragging.
    t.broadcastResize(t.root, 0, 0, cw, ch)
}

// findBorders recursively collects all border positions from the node tree.
func findBorders(node *Node, x, y, w, h int) []BorderHit {
    if node == nil || node.IsLeaf() {
        return nil
    }

    var borders []BorderHit

    if node.Split != nil {
        split := node.Split
        switch split.Direction {
        case Vertical:
            borderW := 1
            availW := w - borderW
            firstW, secondW := computeSplitSizes(...)
            borderX := x + firstW
            // Omit the border when one side is collapsed.
            if !split.First.IsCollapsed() && !split.Second.IsCollapsed() {
                borders = append(borders, BorderHit{
                    Split:     split,
                    Direction: Vertical,
                    X:         borderX,
                    Y:         y,
                    Length:    h,
                })
            }
            // ...
        case Horizontal:
            // Similar logic for horizontal splits
        }
    }

    if node.Flex != nil {
        borders = append(borders, findFlexBorders(node.Flex, x, y, w, h)...)
    }

    return borders
}

type BorderHit struct {

    X, Y      int        // Screen coordinates
